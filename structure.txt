## Database Schema

### Tables

#### 1. `students`
```sql
CREATE TABLE students (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    student_id VARCHAR(50) UNIQUE NOT NULL,  -- School-provided ID
    name VARCHAR(200) NOT NULL,
    age INTEGER,
    gender VARCHAR(10),
    email VARCHAR(255),
    phone VARCHAR(20),
    address TEXT,
    school_id INTEGER REFERENCES schools(id),
    grade_level VARCHAR(50),
    enrollment_date DATE,
    status VARCHAR(20) DEFAULT 'active',  -- active, at_risk, dropped_out, graduated
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 2. `schools`
```sql
CREATE TABLE schools (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(200) NOT NULL,
    code VARCHAR(50) UNIQUE NOT NULL,
    type VARCHAR(50),  -- primary, secondary, college
    address TEXT,
    city VARCHAR(100),
    state VARCHAR(100),
    country VARCHAR(100) DEFAULT 'India',
    contact_email VARCHAR(255),
    contact_phone VARCHAR(20),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 3. `attendance_records`
```sql
CREATE TABLE attendance_records (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    student_id INTEGER REFERENCES students(id),
    date DATE NOT NULL,
    status VARCHAR(20) NOT NULL,  -- present, absent, late, excused
    reason TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(student_id, date)
);
```

#### 4. `fee_records`
```sql
CREATE TABLE fee_records (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    student_id INTEGER REFERENCES students(id),
    amount_due DECIMAL(10, 2) NOT NULL,
    amount_paid DECIMAL(10, 2) DEFAULT 0,
    due_date DATE,
    status VARCHAR(20) DEFAULT 'pending',  -- pending, partial, paid, overdue
    months_overdue INTEGER DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 5. `health_records`
```sql
CREATE TABLE health_records (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    student_id INTEGER REFERENCES students(id),
    issue_type VARCHAR(100),  -- chronic, acute, mental_health
    description TEXT,
    severity VARCHAR(20),  -- low, medium, high, critical
    affects_attendance BOOLEAN DEFAULT FALSE,
    reported_date DATE,
    resolved_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 6. `academic_records`
```sql
CREATE TABLE academic_records (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    student_id INTEGER REFERENCES students(id),
    semester VARCHAR(50),
    subject VARCHAR(100),
    grade DECIMAL(5, 2),
    percentage DECIMAL(5, 2),
    exam_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 7. `conversations` (Chatbot conversations)
```sql
CREATE TABLE conversations (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    student_id INTEGER REFERENCES students(id),
    session_id VARCHAR(100) UNIQUE NOT NULL,
    started_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ended_at TIMESTAMP,
    status VARCHAR(20) DEFAULT 'active',  -- active, resolved, escalated
    language VARCHAR(10) DEFAULT 'en',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 8. `messages` (Chat messages)
```sql
CREATE TABLE messages (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    conversation_id INTEGER REFERENCES conversations(id),
    role VARCHAR(20) NOT NULL,  -- user, assistant, system
    content TEXT NOT NULL,
    problem_category VARCHAR(100),  -- financial, family, health, etc.
    severity VARCHAR(20),
    sentiment VARCHAR(20),  -- positive, neutral, negative
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 9. `risk_assessments`
```sql
CREATE TABLE risk_assessments (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    student_id INTEGER REFERENCES students(id),
    risk_score DECIMAL(5, 2) NOT NULL,  -- 0-100
    risk_level VARCHAR(20) NOT NULL,  -- low, medium, high, critical
    factors JSON,  -- JSON object with risk factors and scores
    calculated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 10. `interventions`
```sql
CREATE TABLE interventions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    student_id INTEGER REFERENCES students(id),
    risk_assessment_id INTEGER REFERENCES risk_assessments(id),
    type VARCHAR(50) NOT NULL,  -- financial_aid, counseling, academic_support, etc.
    status VARCHAR(20) DEFAULT 'pending',  -- pending, in_progress, completed, cancelled
    recommended_resources JSON,  -- Array of resource IDs
    action_plan TEXT,
    assigned_to INTEGER,  -- Counselor/admin ID
    started_at TIMESTAMP,
    completed_at TIMESTAMP,
    outcome TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 11. `resources` (Knowledge base)
```sql
CREATE TABLE resources (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title VARCHAR(200) NOT NULL,
    type VARCHAR(50) NOT NULL,  -- scholarship, program, organization, legal, etc.
    category VARCHAR(100),  -- financial, health, legal, academic
    description TEXT,
    eligibility_criteria TEXT,
    application_process TEXT,
    contact_info JSON,  -- {email, phone, website, address}
    location VARCHAR(200),
    availability VARCHAR(20),  -- available, limited, closed
    tags JSON,  -- Array of tags for search
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### 12. `resource_recommendations`
```sql
CREATE TABLE resource_recommendations (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    student_id INTEGER REFERENCES students(id),
    conversation_id INTEGER REFERENCES conversations(id),
    resource_id INTEGER REFERENCES resources(id),
    recommended_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    student_viewed BOOLEAN DEFAULT FALSE,
    student_applied BOOLEAN DEFAULT FALSE,
    outcome VARCHAR(50)  -- applied, accepted, rejected, pending
);
```

### `models/database.py`
```python
import sqlite3
from contextlib import contextmanager

DB_PATH = "women_education.db"

@contextmanager
def get_connection():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()

def init_db():
    """Initialize database with all tables"""
    with get_connection() as conn:
        # Create all tables
        conn.executescript("""
            CREATE TABLE IF NOT EXISTS students (...);
            CREATE TABLE IF NOT EXISTS schools (...);
            CREATE TABLE IF NOT EXISTS attendance_records (...);
            -- ... all other tables
        """)
```
